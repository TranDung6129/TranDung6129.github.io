{{- /* Page by content path in the current language, else English. The argument may also be a list of alternative paths (D-017): the first found in the current language wins, then the first found in English. Returns nil when none exists. */ -}} {{- $paths := cond (reflect.IsSlice .) . (slice .) -}} {{- $p := false -}} {{- range $paths }}{{ if not $p }}{{ with site.GetPage . }}{{ $p = . }}{{ end }}{{ end }}{{ end -}} {{- if not $p -}} {{- range site.Sites -}} {{- if eq .Language.Lang "en" -}} {{- $s := . -}} {{- range $paths }}{{ if not $p }}{{ with $s.GetPage . }}{{ $p = . }}{{ end }}{{ end }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $p -}}
