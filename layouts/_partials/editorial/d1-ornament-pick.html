{{- /* Deterministic D1 ornament choice (D-016). Returns dict "branch" "toc" "landscape" (file names). Same article → same pick on every build and reload: the choice is an md5 of the content path (language-independent), so translations share their artwork. Adding an asset with a role in data/d1_ornaments.yaml widens the rotation without template changes. Front matter override: ornaments: { branch: "d5/branch-mattioli-vicia.webp", toc: "d5/flower-fuchs-primula.webp" } */ -}} {{- $p := . -}} {{- $key := $p.RelPermalink -}} {{- with $p.File }}{{ $key = printf "%s/%s" .Dir .TranslationBaseName }}{{ end -}} {{- $seed := int (printf "0x%s" (substr (md5 $key) 0 6)) -}} {{- $out := dict -}} {{- $over := $p.Params.ornaments | default dict -}} {{- range $role := slice "branch" "toc" "landscape" -}} {{- $files := slice -}} {{- range site.Data.d1_ornaments.assets }}{{ if eq .role $role }}{{ $files = $files | append .path }}{{ end }}{{ end -}} {{- $pick := "" -}} {{- with index $over $role }}{{ if in $files . }}{{ $pick = . }}{{ else }}{{ warnf "%s: ornament %q is not a %s asset in data/d1_ornaments.yaml" $p.File.Path . $role }}{{ end }}{{ end -}} {{- if and (not $pick) $files }}{{ $pick = index $files (mod (add $seed (index (dict "branch" 2 "toc" 0 "landscape" 1) $role)) (len $files)) }}{{ end -}} {{- $out = merge $out (dict $role $pick) -}} {{- end -}} {{- return $out -}}
